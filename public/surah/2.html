تفسير سورة البقرة عند ابن تيمية وابن القيم

الآيات التي جُمع فيها كلام الشيخين من سورة البقرة (١٣٢ من ٢٨٦ آية). اختر آية:

٢٣٤٥٦٧٨٩١٠١١١٢١٤١٥١٧١٨١٩٢٠٢١٢٢٢٤٢٦٢٨٢٩٣٠٣٦٣٧٣٨٤٢٤٥٥٨٦٢٦٦٧٤٧٨٨١٨٤٨٥٨٨٨٩٩٠٩١٩٧١٠٣١٠٦١٠٩١١١١١٢١١٥١١٦١١٧١١٨١٢١١٢٣١٢٨١٣١١٣٣١٣٥١٣٦١٤٢١٤٣١٤٨١٥١١٥٢١٥٦١٥٨١٥٩١٦٣١٦٥١٦٦١٧١١٧٢١٧٧١٧٨١٧٩١٨٢١٨٤١٨٥١٨٦١٨٧١٨٨١٨٩١٩٠١٩٣١٩٦١٩٧٢٠٠٢٠١٢٠٣٢٠٤٢٠٧٢١٣٢١٤٢١٧٢٢١٢٢٣٢٢٤٢٢٥٢٢٦٢٢٧٢٢٨٢٢٩٢٣٠٢٣١٢٣٢٢٣٣٢٣٥٢٣٦٢٤٥٢٥١٢٥٣٢٥٤٢٥٥٢٥٦٢٥٧٢٦١٢٦٢٢٦٤٢٦٦٢٦٧٢٦٨٢٧١٢٧٢٢٧٣٢٧٤٢٧٥٢٧٦٢٧٨٢٧٩٢٨٢٢٨٤٢٨٥٢٨٦
